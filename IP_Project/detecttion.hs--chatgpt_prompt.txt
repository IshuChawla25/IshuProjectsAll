import React, { useState, useRef, useEffect } from "react";
// import "bootstrap/dist/css/bootstrap.min.css";
import { Navbar, Nav } from "react-bootstrap";
import { Link } from "react-router-dom";
import './detection.css';

const Detection = () => {
  const [image, setImage] = useState(null);
  const [flowerDetails, setFlowerDetails] = useState(null);
  const [cameraActive, setCameraActive] = useState(false);
  const [loading, setLoading] = useState(false);
  const videoRef = useRef(null);
  const canvasRef = useRef(document.createElement("canvas"));
  const streamRef = useRef(null);

  useEffect(() => {
    if (cameraActive) {
      navigator.mediaDevices.getUserMedia({ video: true })
        .then((stream) => {
          streamRef.current = stream;
          if (videoRef.current) {
            videoRef.current.srcObject = stream;
            videoRef.current.onloadedmetadata = () => {
              videoRef.current.play().catch(error => console.error("Video play error:", error));
            };
          }
        })
        .catch((err) => {
          console.error("Error accessing camera:", err);
          alert("Camera access error: " + err.message);
          setCameraActive(false);
        });
    } else {
      if (streamRef.current) {
        streamRef.current.getTracks().forEach(track => track.stop());
        streamRef.current = null;
      }
    }

    return () => {
      if (streamRef.current) {
        streamRef.current.getTracks().forEach(track => track.stop());
        streamRef.current = null;
      }
    };
  }, [cameraActive]);

  const handleImageUpload = async (event) => {
    if (cameraActive) closeCamera();
    const file = event.target.files[0];
    if (!file) return;

    const formData = new FormData();
    formData.append("image", file);

    try {
      setLoading(true);
      const response = await fetch("http://127.0.0.1:5000/detect-flower", {
        method: "POST",
        body: formData
      });
      const data = await response.json();
      setFlowerDetails(data);
      setImage(URL.createObjectURL(file));
    } catch (error) {
      console.error("Error detecting flower:", error);
      alert("Detection failed. Please try again.");
    } finally {
      setLoading(false);
    }
  };

  const captureImage = () => {
    const video = videoRef.current;
    const canvas = canvasRef.current;
    const context = canvas.getContext("2d");
    canvas.width = video.videoWidth;
    canvas.height = video.videoHeight;
    context.drawImage(video, 0, 0, canvas.width, canvas.height);

    canvas.toBlob(async (blob) => {
      const formData = new FormData();
      formData.append("image", blob, "captured.png");
      try {
        setLoading(true);
        const response = await fetch("http://127.0.0.1:5000/detect-flower", {
          method: "POST",
          body: formData
        });
        const data = await response.json();
        setFlowerDetails(data);
        setImage(URL.createObjectURL(blob));
      } catch (error) {
        console.error("Error detecting flower:", error);
        alert("Capture failed. Please try again.");
      } finally {
        setLoading(false);
      }
    }, "image/png");
    closeCamera();
  };

  const closeCamera = () => {
    if (streamRef.current) {
      streamRef.current.getTracks().forEach(track => track.stop());
      streamRef.current = null;
    }
    setCameraActive(false);
  };

  const resetApp = () => {
    setImage(null);
    setFlowerDetails(null);
    setLoading(false);
    closeCamera();
  };
  const knowMore = () => {
    if (flowerDetails?.name) {
      const query = `Tell me more about the ${flowerDetails.name} flower.`;
      window.open(`https://chatgpt.com/?prompt=${encodeURIComponent(query)}`);
    }
  };
  
  return (
    <div className="gradient-background text-dark">
      {/* Navbar */}
      <Navbar bg="white" expand="lg" className="shadow-sm border-bottom py-3">
        <div className="container">
          <Navbar.Brand href="#">
            <img src="https://storage.googleapis.com/a1aa/image/32eb0098-4ed9-4866-182a-c4d62cb0c465.jpg" alt="Logo" width="50" className="me-2" />
            <strong className="text-success fs-4">PetalVision</strong>
          </Navbar.Brand>
          <Navbar.Toggle />
          <Navbar.Collapse>
            <Nav className="ms-auto" style={{ fontSize: "20px" }}>
              <Nav.Link as={Link} to="/home" className="text-success fw-bold">Home</Nav.Link>
              <Nav.Link as={Link} to="/features" className="text-success fw-bold">Features</Nav.Link>
              <Nav.Link as={Link} to="/detection" className="text-success fw-bold">Detection</Nav.Link>
              <Nav.Link as={Link} to="/about" className="text-success fw-bold">About</Nav.Link>
              <Nav.Link href="/login_signup" className="text-success fw-bold">Login</Nav.Link>
            </Nav>
          </Navbar.Collapse>
        </div>
      </Navbar>

      {/* Main Section */}
      <main className="container flex-grow-1 text-center py-5">
        <h1 className="fs-4xl fw-bold text-green-800 mb-4">Welcome to PetalVision</h1>
        <p className="text-green-700 fs-5 mb-4">Scan or upload any flower and let AI tell you all about it!</p>

        <div className="mb-4 d-flex justify-content-center gap-3 flex-wrap">
          <input type="file" id="imageUpload" accept="image/*" className="d-none" onChange={handleImageUpload} />
          <button className="btn btn-primary px-4" onClick={() => document.getElementById("imageUpload").click()} disabled={loading}>
            Upload Image
          </button>
          <button className="btn btn-success px-4" onClick={() => { setImage(null); setFlowerDetails(null); setCameraActive(true); }} disabled={loading}>
            Open Camera
          </button>
          <button className="btn btn-danger px-4" onClick={resetApp}>
            Reset
          </button>
        </div>

        {cameraActive && (
          <div className="border rounded p-3 bg-white mx-auto" style={{ maxWidth: "520px" }}>
            <h5 className="text-info">Live Camera Preview</h5>
            <video ref={videoRef} autoPlay playsInline muted className="w-100 rounded border"></video>
            <div className="mt-3 d-flex justify-content-center gap-3">
              <button className="btn btn-info" onClick={captureImage}>Capture</button>
              <button className="btn btn-outline-danger" onClick={closeCamera}>Close</button>
            </div>
          </div>
        )}

        {loading && <p className="text-warning mt-3">Detecting flower...</p>}
        {image && <img src={image} alt="Detected Flower" className="img-fluid rounded shadow mt-4" style={{ maxWidth: "500px" }} />}

        {flowerDetails && (
          <div className="card shadow mx-auto mt-4 text-start" style={{ maxWidth: "500px" }}>
            <div className="card-body">
              <h3 className="card-title text-success">{flowerDetails.name}</h3>
              <p className="card-subtitle mb-2 text-muted">{flowerDetails.season}</p>
              <p className="card-text">{flowerDetails.description}</p>
              <div className="text-center">
                <button className="btn btn-info" onClick={knowMore}>Know More</button>
              </div>
            </div>
          </div>
        )}
      </main>

      {/* Footer */}
      <footer className="bg-success text-white pt-4 pb-3 mt-auto">
        <div className="container row mx-auto">
          <div className="col-md-4 mb-3">
            <h5>PetalVision</h5>
            <p>Explore and identify beautiful flowers with AI-powered insights.</p>
          </div>
          <div className="col-md-4 mb-3">
            <h5>Quick Links</h5>
            <ul className="list-unstyled">
                <li><Nav.Link href="/" className="text-white text-decoration-none">Home</Nav.Link></li>
                <li><Nav.Link href="/features" className="text-white text-decoration-none">Features</Nav.Link></li>
                <li><Nav.Link href="/detection" className="text-white text-decoration-none">Detection</Nav.Link></li>
                <li><Nav.Link href="/about" className="text-white text-decoration-none">About</Nav.Link></li>
                <li><a href="#contact" className="text-white text-decoration-none">Contact</a></li>
            </ul>
          </div>
          <div className="col-md-4 mb-3">
            <h5>Contact Us</h5>
            <address>
              123 Blossom Lane, Garden City, FL<br />
              +1 (555) 123-4567<br />
              support@petalvision.com
            </address>
          </div>
        </div>
        <div className="text-center border-top border-white mt-3 pt-2 small">
          © 2025 PetalVision. All rights reserved.
        </div>
      </footer>
    </div>
  );
};

export default Detection;
